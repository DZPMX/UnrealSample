def instance_proof(graph,member,is_before):
    ns,es,cs,_=graph_model(graph)
    @functools.lru_cache(None)
    def sig(i,index=0):
        n=ns[i];kind=n['class'];name=n.get('parameter_name')
        if kind=='StaticSwitchParameter':
            value=(old_values[member]['switch'] if is_before else expected[member]['switch'])[name]
            s,o=es[i]['True' if value else 'False'];return sig(s,o)
        if kind=='ScalarParameter':return digest(['Scalar',*(translation.get(name,(name,0)) if is_before else (name,0))])
        if kind=='VectorParameter':return digest(['Scalar',name,index])
        props={k:v for k,v in n['properties'].items() if k not in ['Desc','SortPriority','Name','NodeColor','Texture']};props.update(n.get('precise',{}))
        target=translation.get(name,(name,0))[0] if is_before else name
        return digest([kind,target,index,props,[(pin,sig(s,o)) for pin,(s,o) in sorted(es[i].items())]])
    return [sig(i) for i,n in ns.items() if n['object_path']==graph['attributes_root']]
